182 FUNC length distance_on_sphere_formula(angle lat1, angle long1,
183 angle lat2, angle long2,
184 length radius = 6371.0088_km) // mean Earth radius, recommended by NOAA
185 {
186 angle dLat = lat1 - lat2; // difference in latitude
187 angle dLong = long1 - long2; // difference in longitude
188
189 angle a = sin2(dLat / 2.) + cos(lat1) * cos(lat2) * sin2(dLong / 2.); // Haversine intermediate
190 auto c = 2. * atan2(sqrt(a), sqrt(1. - a)); // central angle
191
192 return c * radius;
193 }
