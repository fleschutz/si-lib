21 static bool _parse(const std::string& str, double& number, char* unit) {
22#ifdef _WIN32
23 return sscanf_s(str.c_str(), "%lf%99s", &number, unit, 100) == 2;
24#else
25 return std::sscanf(str.c_str(), "%lf%99s", &number, unit) == 2;
26#endif
27 }

30 FUNC length to_length(const std::string& str) {
31 double num = 0.0;
32 char unit[100] = "";
33 if (!_parse(str, num, unit)) return meters(std::nan("1")); // not recognized
34 if (!std::strcmp(unit, "Gm")) return num * 1_Gm;
35 if (!std::strcmp(unit, "Mm")) return num * 1_Mm;
36 if (!std::strcmp(unit, "km")) return num * 1_km;
37 if (!std::strcmp(unit, "m")) return num * 1_m;
38 if (!std::strcmp(unit, "dm")) return num * 1_dm;
39 if (!std::strcmp(unit, "cm")) return num * 1_cm;
40 if (!std::strcmp(unit, "mm")) return num * 1_mm;
41 if (!std::strcmp(unit, "um")) return num * 1_um;
42 if (!std::strcmp(unit, "nm")) return num * 1_nm;
43 if (!std::strcmp(unit, "pm")) return num * 1_pm;
44 return meters(std::nan("2")); // unknown unit
45 }
