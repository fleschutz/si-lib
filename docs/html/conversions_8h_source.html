23 static bool _parse(const std::string& str, double& number, char* unit) {
24#ifdef _WIN32
25 return sscanf_s(str.c_str(), "%lf%99s", &number, unit, 100) == 2;
26#else
27 return std::sscanf(str.c_str(), "%lf%99s", &number, unit) == 2;
28#endif
29 }

32 FUNC length to_length(const std::string& str) {
33 double num = 0.0;
34 char unit[100] = "";
35 if (!_parse(str, num, unit)) return meters(std::nan("1")); // not recognized
36 if (!std::strcmp(unit, "Gm")) return num * 1_Gm;
37 if (!std::strcmp(unit, "Mm")) return num * 1_Mm;
38 if (!std::strcmp(unit, "km")) return num * 1_km;
39 if (!std::strcmp(unit, "m")) return num * 1_m;
40 if (!std::strcmp(unit, "dm")) return num * 1_dm;
41 if (!std::strcmp(unit, "cm")) return num * 1_cm;
42 if (!std::strcmp(unit, "mm")) return num * 1_mm;
43 if (!std::strcmp(unit, "um")) return num * 1_um;
44 if (!std::strcmp(unit, "nm")) return num * 1_nm;
45 if (!std::strcmp(unit, "pm")) return num * 1_pm;
46 return meters(std::nan("2")); // unknown unit
47 }
