23 FUNC bool from_string(const std::string& str, length& result) {
24 double value = 0.0;
25 char unit[100] = "";
26 if (std::sscanf(str.c_str(), "%lf%99s", &value, unit) != 2)
27 return false; // not recognized
28
29 if (!std::strcmp(unit, "Gm")) result = 1_Gm * value;
30 else if (!std::strcmp(unit, "Mm")) result = 1_Mm * value;
31 else if (!std::strcmp(unit, "km")) result = 1_km * value;
32 else if (!std::strcmp(unit, "m")) result = 1_m * value;
33 else if (!std::strcmp(unit, "dm")) result = 1_dm * value;
34 else if (!std::strcmp(unit, "cm")) result = 1_cm * value;
35 else if (!std::strcmp(unit, "mm")) result = 1_mm * value;
36 else if (!std::strcmp(unit, "um")) result = 1_um * value;
37 else if (!std::strcmp(unit, "nm")) result = 1_nm * value;
38 else if (!std::strcmp(unit, "pm")) result = 1_pm * value;
39 else
40 return false; // unknown unit
41
42 return true;
43 }
